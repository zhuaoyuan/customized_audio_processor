# audio_processer 运行依赖（Python 3.10 / Windows）

# 安装（本项目实测命令）：
#   $env:PIP_CACHE_DIR = "$PWD\work\pip_cache"      # 必须！见下方「坑」
#   work\venv\Scripts\python -m pip install -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple
#
# 坑（2026-10-21 实测）：本机全局 pip 缓存 %LOCALAPPDATA%\pip\cache 有 3.8 GB / 1000+ 文件，
#   `pip install` 一旦读写它会**无输出空转**（CPU 占用拉满、10 min 不结束），
#   而 PyPI 直连又超时。把 PIP_CACHE_DIR 指到一个新目录（或加 --no-cache-dir）即可 12 s 装完。
#   详见 docs/工程日志.md「环境坑」。
#
# 版本 = 本机实测通过的组合，不要随意漂移。
